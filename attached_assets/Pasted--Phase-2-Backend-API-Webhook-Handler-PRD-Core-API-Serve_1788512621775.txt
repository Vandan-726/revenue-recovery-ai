## 4. DATABASE SCHEMA

### 4.1 Events Table
```sql
CREATE TABLE events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    event_type VARCHAR(50) NOT NULL,  -- payment.failed, payment.authorized, order.paid
    customer_id VARCHAR(50),
    payment_id VARCHAR(50),
    order_id VARCHAR(50),
    amount BIGINT,  -- in paise
    currency VARCHAR(3) DEFAULT 'INR',
    error_code VARCHAR(100),
    error_description TEXT,
    metadata JSONB,  -- Store complete webhook payload
    status VARCHAR(20) DEFAULT 'pending',  -- pending, processed, failed
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    processed_at TIMESTAMP
);

CREATE INDEX idx_events_payment_id ON events(payment_id);
CREATE INDEX idx_events_customer_id ON events(customer_id);
CREATE INDEX idx_events_status ON events(status);
CREATE INDEX idx_events_created_at ON events(created_at DESC);
```

### 4.2 Payments Table
```sql
CREATE TABLE payments (
    id VARCHAR(50) PRIMARY KEY,  -- Razorpay payment_id
    order_id VARCHAR(50),
    customer_id VARCHAR(50),
    amount BIGINT,  -- in paise
    currency VARCHAR(3) DEFAULT 'INR',
    status VARCHAR(20),  -- created, authorized, captured, failed
    method VARCHAR(20),  -- card, netbanking, wallet, emi
    error_code VARCHAR(100),
    error_description TEXT,
    razorpay_response JSONB,  -- Store complete response
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_payments_customer_id ON payments(customer_id);
CREATE INDEX idx_payments_status ON payments(status);
```

### 4.3 Recoveries Table
```sql
CREATE TABLE recoveries (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    payment_id VARCHAR(50) NOT NULL REFERENCES payments(id),
    event_id UUID REFERENCES events(id),
    customer_id VARCHAR(50),
    amount BIGINT,
    root_cause VARCHAR(255),  -- Why it failed
    status VARCHAR(20) DEFAULT 'pending',  -- pending, in_progress, success, failed
    strategies JSONB,  -- Array of recovery strategies
    selected_strategy VARCHAR(50),  -- Current active strategy
    max_attempts INT DEFAULT 3,
    attempts_used INT DEFAULT 0,
    recovered_at TIMESTAMP,
    recovery_amount BIGINT,  -- Amount actually recovered
    revenue_impact BIGINT,  -- Net revenue gain/loss
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_recoveries_payment_id ON recoveries(payment_id);
CREATE INDEX idx_recoveries_customer_id ON recoveries(customer_id);
CREATE INDEX idx_recoveries_status ON recoveries(status);
```

### 4.4 Recovery Attempts Table
```sql
CREATE TABLE recovery_attempts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    recovery_id UUID NOT NULL REFERENCES recoveries(id),
    payment_id VARCHAR(50),
    action_type VARCHAR(50),  -- retry, sms, email, discount, call
    status VARCHAR(20),  -- pending, executed, success, failed
    attempt_number INT,
    scheduled_for TIMESTAMP,
    executed_at TIMESTAMP,
    result JSONB,  -- Response from action
    error_message TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_recovery_attempts_recovery_id ON recovery_attempts(recovery_id);
CREATE INDEX idx_recovery_attempts_status ON recovery_attempts(status);
```

### 4.5 Audit Logs Table

```python
# app/api/v1/endpoints/webhooks.py
from fastapi import APIRouter, Request, HTTPException, status
from app.services.webhook_handler import handle_razorpay_webhook
import hmac
import hashlib
import json

router = APIRouter()

@router.post("/razorpay")
async def razorpay_webhook(request: Request):
    """
    Handle Razorpay webhook events
    
    Validates signature, processes event, stores in DB
    """
    body = await request.body()
    signature = request.headers.get("X-Razorpay-Signature")
    
    # Validate signature
    expected_signature = hmac.new(
        settings.RAZORPAY_WEBHOOK_SECRET.encode(),
        body,
        hashlib.sha256
    ).hexdigest()
    
    if signature != expected_signature:
        raise HTTPException(status_code=401, detail="Invalid signature")
    
    data = json.loads(body)
    
    try:
        result = await handle_razorpay_webhook(data)
        return {
            "status": "ok",
            "event_id": result.get("event_id"),
            "message": "Webhook processed successfully"
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```

```python
# app/api/v1/endpoints/recoveries.py
@router.get("", response_model=RecoveryListResponse)
async def list_recoveries(
    page: int = 1,
    per_page: int = 10,
    status: Optional[str] = None,
    search: Optional[str] = None,
    db: Session = Depends(get_db)
):
    """
    List all recoveries with filters
    
    Query Parameters:
    - page: Page number (default: 1)
    - per_page: Items per page (default: 10)
    - status: Filter by status (pending, in_progress, success, failed)
    - search: Search by payment_id or customer_id
    """
    offset = (page - 1) * per_page
    
    query = db.query(Recovery)
    
    if status:
        query = query.filter(Recovery.status == status)
    
    if search:
        query = query.filter(
            (Recovery.payment_id.ilike(f"%{search}%")) |
            (Recovery.customer_id.ilike(f"%{search}%"))
        )
    
    total = query.count()
    items = query.offset(offset).limit(per_page).all()
    
    return {
        "items": items,
        "total": total,
        "page": page,
        "per_page": per_page,
        "total_pages": (total + per_page - 1) // per_page
    }
```

```python
@router.get("/{recovery_id}", response_model=RecoveryDetailResponse)
async def get_recovery_detail(
    recovery_id: str,
    db: Session = Depends(get_db)
):
    """
    Get detailed info about a specific recovery
    """
    recovery = db.query(Recovery).filter(Recovery.id == recovery_id).first()
    
    if not recovery:
        raise HTTPException(status_code=404, detail="Recovery not found")
    
    # Get timeline events
    timeline = db.query(RecoveryAttempt).filter(
        RecoveryAttempt.recovery_id == recovery_id
    ).order_by(RecoveryAttempt.created_at).all()
    
    # Get audit logs
    audit_logs = db.query(AuditLog).filter(
        AuditLog.recovery_id == recovery_id
    ).order_by(AuditLog.created_at).all()
    
    return {
        **recovery.__dict__,
        "timeline": timeline,
        "audit_logs": audit_logs
    }
```

## 8. WEBHOOK HANDLER SERVICE

### 8.1 Webhook Handler
```python
# app/services/webhook_handler.py
from datetime import datetime
from sqlalchemy.orm import Session
from app.models import Event, Payment, Recovery, AuditLog
from app.utils.logger import logger
from app.utils.validators import validate_webhook
import json

class WebhookHandler:
    
    async def handle_payment_failed(
        self,
        data: dict,
        db: Session
    ) -> dict:
        """
        Handle payment.failed event
        
        1. Extract payment details
        2. Store event in DB
        3. Log audit trail
        4. Trigger recovery (queued for Phase 3)
        """
        
        payment_data = data['payload']['payment']['entity']
        
        # 1. Create/Update payment record
        payment = Payment(
            id=payment_data['id'],
            order_id=payment_data.get('order_id'),
            customer_id=payment_data.get('customer_id'),
            amount=payment_data['amount'],
            status=payment_data['status'],
            error_code=payment_data.get('error_code'),
            error_description=payment_data.get('error_description'),
            razorpay_response=payment_data
        )
        db.add(payment)
        db.flush()
        
        # 2. Create event record
        event = Event(
            event_type='payment.failed',
            payment_id=payment_data['id'],
            customer_id=payment_data.get('customer_id'),
            amount=payment_data['amount'],
            error_code=payment_data.get('error_code'),
            error_description=payment_data.get('error_description'),
            metadata=payment_data,
            status='pending'
        )
        db.add(event)
        db.flush()
        
        # 3. Create initial recovery record
        recovery = Recovery(
            payment_id=payment_data['id'],
            event_id=event.id,
            customer_id=payment_data.get('customer_id'),
            amount=payment_data['amount'],
            status='pending'
        )
        db.add(recovery)
        db.flush()
        
        # 4. Log audit trail
        audit_log = AuditLog(
            payment_id=payment_data['id'],
            recovery_id=recovery.id,
            action='Payment failed',
            actor='System',
            details={
                'error_code': payment_data.get('error_code'),
                'error_description': payment_data.get('error_description')
            }
        )
        db.add(audit_log)
        
        db.commit()
        
        logger.info(f"Payment {payment_data['id']} failed, recovery initiated")
        
        return {
            "event_id": event.id,
            "recovery_id": recovery.id,
            "status": "success"
        }
    
    async def handle_order_paid(self, data: dict, db: Session) -> dict:
        """Handle order.paid event - payment successful"""
        order_data = data['payload']['order']['entity']
        
        # Update payment status
        payment = db.query(Payment).filter(
            Payment.id == order_data.get('receipt')
        ).first()
        
        if payment:
            payment.status = 'captured'
            db.commit()
        
        return {"status": "success"}

async def handle_razorpay_webhook(data: dict, db: Session) -> dict:
    """Main webhook handler dispatcher"""
    
    event_type = data.get('event')
    handler = WebhookHandler()
    
    if event_type == 'payment.failed':
        return await handler.handle_payment_failed(data, db)
    elif event_type == 'order.paid':
        return await handler.handle_order_paid(data, db)
    else:
        logger.warning(f"Unhandled event type: {event_type}")
        return {"status": "ignored"}
```

```python
# app/middleware/error_handler.py
from fastapi.responses import JSONResponse
from fastapi import Request
from app.utils.logger import logger

async def global_exception_handler(request: Request, exc: Exception):
    logger.error(f"Exception: {str(exc)}", exc_info=True)
    
    return JSONResponse(
        status_code=500,
        content={
            "success": False,
            "error": "Internal server error",
            "message": str(exc) if settings.DEBUG else "An error occurred"
        }
    )
```
