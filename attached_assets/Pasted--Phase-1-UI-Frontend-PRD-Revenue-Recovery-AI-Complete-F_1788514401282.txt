```typescript
// components/ui/Button.tsx
interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'danger' | 'ghost';
  size?: 'sm' | 'md' | 'lg';
  isLoading?: boolean;
  icon?: React.ReactNode;
}

export const Button: React.FC<ButtonProps> = ({
  variant = 'primary',
  size = 'md',
  isLoading,
  icon,
  children,
  ...props
}) => {
  const baseStyles = 'font-medium rounded-lg transition-all duration-200 flex items-center gap-2';
  
  const variants = {
    primary: 'bg-primary-600 text-white hover:bg-primary-700 active:scale-95',
    secondary: 'bg-neutral-200 text-neutral-900 hover:bg-neutral-300',
    danger: 'bg-danger text-white hover:bg-red-600',
    ghost: 'text-primary-600 hover:bg-primary-50',
  };
  
  const sizes = {
    sm: 'px-3 py-2 text-sm',
    md: 'px-4 py-2.5 text-base',
    lg: 'px-6 py-3 text-lg',
  };
  
  return (
    <button
      className={`${baseStyles} ${variants[variant]} ${sizes[size]}`}
      disabled={isLoading}
      {...props}
    >
      {isLoading && <Spinner size="sm" />}
      {icon && icon}
      {children}
    </button>
  );
};
```

```typescript
// components/ui/Alert.tsx
export const Alert: React.FC<{
  type: 'success' | 'error' | 'warning' | 'info';
  title: string;
  message: string;
  onClose?: () => void;
}> = ({ type, title, message, onClose }) => {
  const icons = {
    success: <CheckCircleIcon />,
    error: <XCircleIcon />,
    warning: <AlertTriangleIcon />,
    info: <InfoIcon />,
  };
  
  const colors = {
    success: 'bg-green-50 border-green-200 text-green-900',
    error: 'bg-red-50 border-red-200 text-red-900',
    warning: 'bg-yellow-50 border-yellow-200 text-yellow-900',
    info: 'bg-blue-50 border-blue-200 text-blue-900',
  };
  
  return (
    <div className={`border rounded-lg p-4 flex gap-3 ${colors[type]}`}>
      <div className="flex-shrink-0">{icons[type]}</div>
      <div className="flex-1">
        <h4 className="font-semibold">{title}</h4>
        <p className="text-sm mt-1">{message}</p>
      </div>
      {onClose && (
        <button onClick={onClose} className="text-lg font-bold">
          ×
        </button>
      )}
    </div>
  );
};

// components/CountUpNumber.tsx
import { useEffect } from 'react';
import { useMotionValue, useTransform, animate } from 'framer-motion';
import { motion } from 'framer-motion';

export const CountUpNumber: React.FC<{ value: number; duration?: number }> = ({
  value,
  duration = 2,
}) => {
  const motionValue = useMotionValue(0);
  const roundedValue = useTransform(motionValue, Math.round);
  
  useEffect(() => {
    const animation = animate(motionValue, value, { duration });
    return animation.stop;
  }, [value]);
  
  return <motion.span>{roundedValue}</motion.span>;
};

// Usage
<CountUpNumber value={1240} duration={2} />
```
